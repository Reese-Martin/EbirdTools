cmake_minimum_required(VERSION 3.20)
project(bigyear_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(BIGYEAR_ORTOOLS "Build the reduced CP-SAT benchmark" OFF)
find_package(Threads REQUIRED)
add_executable(bigyear-engine search.cpp)
target_link_libraries(bigyear-engine PRIVATE Threads::Threads)
if(BIGYEAR_ORTOOLS)
  find_package(ortools CONFIG REQUIRED)
  target_sources(bigyear-engine PRIVATE exact.cpp)
  target_compile_definitions(bigyear-engine PRIVATE BIGYEAR_ORTOOLS)
  target_link_libraries(bigyear-engine PRIVATE ortools::ortools)
endif()
